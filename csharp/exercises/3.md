```csharp
enum TodoState {
    New,
    Pending,
    Done,
    Cancelled,
    Blocked
}

class TodoItem {
    public int Id { get; set; }
    public string Title { get; set; }
    public string Description { get; set; }
    public TodoState State { get; set; } = TodoState.New;
}
```
